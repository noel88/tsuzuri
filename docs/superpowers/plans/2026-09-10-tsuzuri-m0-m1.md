# Tsuzuri M0–M1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 포메라 DM250에서 네트워크 없이 완결되는 번역 작문 드릴 루프를 만든다 — 출제 → 작문 입력 → 형태소 분석 → 참조 제시 → 저장.

**Architecture:** 단일 Go 바이너리. kagome(pure Go)로 일본어·한국어 형태소 분석. 데이터는 append-only JSON Lines. 분석 엔진은 HTTP 클라이언트를 알지 못한다 — M1에는 네트워크 코드가 한 줄도 없다.

**Tech Stack:** Go 1.24+, `github.com/ikawaha/kagome/v2/tokenizer`, `github.com/ikawaha/kagome-dict/ipa`, `github.com/ikawaha/kagome-dict-ko`. 표준 라이브러리 외 다른 의존성 없음.

**Spec:** `docs/superpowers/specs/2026-09-10-tsuzuri-design.md`

## Global Constraints

모든 작업의 요구사항에 암묵적으로 포함된다.

- **CGO_ENABLED=0 고정.** cgo를 요구하는 의존성은 어떤 이유로도 추가하지 않는다. armhf 크로스컴파일이 깨진다.
- **타겟 빌드:** `GOOS=linux GOARCH=arm GOARM=7 CGO_ENABLED=0 go build -o tsuzuri-armv7`
- **개발·테스트는 맥에서 한다.** kagome는 양쪽에서 동작하므로 모든 테스트가 `go test ./...`로 맥에서 돌아야 한다.
- **점수·정오 판정을 데이터 모델에 두지 않는다** (스펙 §5.4). `score`, `correct`, `passed`, `accuracy` 같은 필드를 만들지 않는다. 분석 결과는 사실의 나열이지 판정이 아니다.
- **모든 쓰기는 append-only + `Sync()`.** 기존 줄을 수정하거나 파일을 재작성하지 않는다.
- **한국어에는 BaseForm이 없다.** mecab-ko-dic은 원형을 제공하지 않는다. 대신 `Expression`(feature index 7) 필드를 파싱한다 — Task 3 참조.
- **한 팩은 한 방향으로 고정.** 사전은 항상 하나만 상주한다.
- 커밋 메시지는 한국어. 각 커밋 끝에 다음 두 줄을 붙인다:
  ```
  Co-Authored-By: Claude Opus 5 (1M context) <noreply@anthropic.com>
  ```
- 브랜치: `develop`에서 `feature/<task-slug>`로 분기하고 `develop`으로 PR. `main` 직접 푸시 금지 (pre-push 훅이 막는다).

```go
// internal/pack/pack.go
package pack

import (
	"bufio"
	"encoding/json"
	"fmt"
	"os"
	"path/filepath"
	"strings"
)

// Direction은 문제의 출제 방향이다. 사용자가 작문하는 언어는 이것의 반대편이다.
type Direction string

const (
	// KoToJa: 한국어 제시문 → 일본어로 작문
	KoToJa Direction = "ko2ja"
	// JaToKo: 일본어 제시문 → 한국어로 작문
	JaToKo Direction = "ja2ko"
)

// Style은 문제가 요구하는 문체다.
type Style string

const (
	StylePlain  Style = "plain"  // 보통체 (だ·である / 한다체)
	StylePolite Style = "polite" // 정중체 (です·ます / 해요체·합니다체)
)

// Problem은 문제 하나다. KeyPoints와 Traps는 팩 생성 시점에 함께 받아둔
// 채점 근거이며, 오프라인 분석이 이것에 의존한다.
type Problem struct {
	ID        string    `json:"id"`
	Dir       Direction `json:"dir"`
	Level     string    `json:"level"`
	Topic     string    `json:"topic"`
	Prompt    string    `json:"prompt"`
	Reference string    `json:"reference"`
	KeyPoints []string  `json:"key_points"`
	Traps     []string  `json:"traps"`
	Style     Style     `json:"style"`
}

// Load는 JSON Lines 팩 파일 하나를 읽는다. 빈 줄은 건너뛴다.
func Load(path string) ([]Problem, error) {
	f, err := os.Open(path)
	if err != nil {
		return nil, err
	}
	defer f.Close()

	var out []Problem
	sc := bufio.NewScanner(f)
	// 기본 버퍼는 64KB다. 긴 문장이 들어갈 수 있으니 넉넉히 잡는다.
	sc.Buffer(make([]byte, 0, 64*1024), 1024*1024)

	for line := 1; sc.Scan(); line++ {
		text := strings.TrimSpace(sc.Text())
		if text == "" {
			continue
		}
		var p Problem
		if err := json.Unmarshal([]byte(text), &p); err != nil {
			return nil, fmt.Errorf("%s %d번째 줄: %w", path, line, err)
		}
		out = append(out, p)
	}
	if err := sc.Err(); err != nil {
		return nil, err
	}
	return out, nil
}

// LoadDir은 디렉터리의 모든 .jsonl을 읽어 방향이 일치하는 문제만 돌려준다.
func LoadDir(dir string, d Direction) ([]Problem, error) {
	paths, err := filepath.Glob(filepath.Join(dir, "*.jsonl"))
	if err != nil {
		return nil, err
	}
	var out []Problem
	for _, p := range paths {
		ps, err := Load(p)
		if err != nil {
			return nil, err
		}
		for _, pr := range ps {
			if pr.Dir == d {
				out = append(out, pr)
			}
		}
	}
	return out, nil
}
```

```go
// internal/store/store_test.go
package store

import (
	"path/filepath"
	"testing"
	"time"
)

func TestAppendThenReadAll(t *testing.T) {
	path := filepath.Join(t.TempDir(), "attempts.jsonl")

	a1 := Attempt{ID: "a001", PackID: "p001", At: time.Now(), Answer: "первый"}
	a2 := Attempt{ID: "a002", PackID: "p002", At: time.Now(), Answer: "두번째"}

	if err := Append(path, a1); err != nil {
		t.Fatalf("첫 Append: %v", err)
	}
	if err := Append(path, a2); err != nil {
		t.Fatalf("둘째 Append: %v", err)
	}

	got, err := ReadAll[Attempt](path)
	if err != nil {
		t.Fatalf("ReadAll: %v", err)
	}
	if len(got) != 2 {
		t.Fatalf("개수 = %d, 기대 2", len(got))
	}
	if got[0].ID != "a001" || got[1].ID != "a002" {
		t.Errorf("순서가 보존되어야 한다: %q, %q", got[0].ID, got[1].ID)
	}
	if got[1].Answer != "두번째" {
		t.Errorf("Answer = %q", got[1].Answer)
	}
}

func TestReadAllTolerantOfTruncatedLastLine(t *testing.T) {
	// 포메라가 갑자기 꺼져 마지막 줄이 잘린 상황을 흉내낸다.
	path := filepath.Join(t.TempDir(), "attempts.jsonl")

		t.Fatal(err)
	}
	if err := appendRaw(path, `{"id":"a002","pack_i`); err != nil {
		t.Fatal(err)
	}

	got, err := ReadAll[Attempt](path)
	if err != nil {
		t.Fatalf("잘린 마지막 줄은 무시되어야 한다: %v", err)
```

```go
package store

import (
	"bufio"
	"encoding/json"
	"errors"
	"io/fs"
	"os"
	"strings"
	"time"

// Attempt는 사용자가 제출한 답안 하나다.
// Analysis는 any다 — 저장소는 분석 결과의 형태를 알 필요가 없다.
type Attempt struct {
	ID       string    `json:"id"`
	PackID   string    `json:"pack_id"`
	At       time.Time `json:"at"`
	Answer   string    `json:"answer"`
	Analysis any       `json:"analysis,omitempty"`
}

// QueueItem은 첨삭 대기 항목이다.
type QueueItem struct {
	AttemptID string    `json:"attempt_id"`
	Priority  bool      `json:"priority"`
	At        time.Time `json:"at"`
}

// Append는 v를 JSON 한 줄로 파일 끝에 덧붙이고 디스크에 flush한다.
// 기존 내용은 절대 수정하지 않는다.
func Append(path string, v any) error {
	b, err := json.Marshal(v)
	if err != nil {
		return err
	}
	return appendRaw(path, string(b))
}

func appendRaw(path, line string) error {
	f, err := os.OpenFile(path, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0o644)
	if err != nil {
		return err
	}
	defer f.Close()

	if _, err := f.WriteString(line + "\n"); err != nil {
		return err
	}
	// 포메라는 예고 없이 꺼진다. 매 쓰기마다 디스크까지 내린다.
	return f.Sync()
}

// ReadAll은 JSON Lines 파일 전체를 읽는다.
// 파일이 없으면 빈 슬라이스를 돌려준다.
// 마지막 줄이 잘려 있으면(전원 차단) 그 줄만 버린다.
func ReadAll[T any](path string) ([]T, error) {
	f, err := os.Open(path)
	if err != nil {
		if errors.Is(err, fs.ErrNotExist) {
			return nil, nil
		}
		return nil, err
	}
	defer f.Close()

	var out []T
	sc := bufio.NewScanner(f)
	sc.Buffer(make([]byte, 0, 64*1024), 1024*1024)

	var pending string
	var pendingOK bool
	for sc.Scan() {
		text := strings.TrimSpace(sc.Text())
		if text == "" {
			continue
		}
		// 직전 줄을 확정한다. 마지막 줄만 손상 가능성이 있으므로
		// 한 줄 늦게 파싱하는 대신, 여기서는 즉시 파싱하되
		// 실패한 줄이 마지막인지로 판단한다.
		var v T
		if err := json.Unmarshal([]byte(text), &v); err != nil {
			// 손상된 줄. 이것이 마지막 줄이면 무시, 아니면 오류.
			pending, pendingOK = text, false
			continue
		}
		if pending != "" && !pendingOK {
			return nil, errors.New("중간에 손상된 줄이 있다: " + pending)
		}
		out = append(out, v)
	}
	if err := sc.Err(); err != nil {
		return nil, err
	}
	return out, nil
}
```

```go
// internal/analyze/token.go
package analyze

import (
	"fmt"
	"strings"

	koDict "github.com/ikawaha/kagome-dict-ko"
	"github.com/ikawaha/kagome-dict/ipa"
	"github.com/ikawaha/kagome/v2/tokenizer"

	"github.com/noel88/tsuzuri/internal/pack"
)

// Token은 사전 종류에 무관한 형태소 표현이다.
// 일본어·한국어의 feature 구조 차이는 전부 이 파일 안에서 흡수한다.
type Token struct {
	Surface string
	Base    string // 기본형. 한국어는 Expression에서 뽑는다.
	POS     string // 품사 대분류 (일본어: 名詞/動詞…, 한국어: NNG/VV…)
	Unknown bool   // 사전에 없는 토큰 — 오타·활용 오류의 신호
}

// IsContent는 이 토큰이 내용어(명사·동사·형용사·부사)인지 판단한다.
// 어휘 비교에서 조사·어미를 제외하기 위해 쓴다.
func (t Token) IsContent() bool {
	switch {
	case t.POS == "名詞", t.POS == "動詞", t.POS == "形容詞", t.POS == "副詞":
		return true
	}
	// 한국어(mecab-ko-dic) 태그. 복합 태그는 첫 성분으로 판단한다.
	head := t.POS
	if i := strings.Index(head, "+"); i >= 0 {
		head = head[:i]
	}
	switch {
	case strings.HasPrefix(head, "NN"), // 명사류
		strings.HasPrefix(head, "VV"), // 동사
		strings.HasPrefix(head, "VA"), // 형용사
		strings.HasPrefix(head, "MAG"): // 일반부사
		return true
	}
	return false
}

// Tokenizer는 문장을 형태소로 나눈다.
type Tokenizer interface {
	Tokenize(string) []Token
}

// NewTokenizer는 사용자가 **작문하는 언어**의 사전을 연다.
// 제시문 언어가 아니라는 점에 주의한다.
func NewTokenizer(d pack.Direction) (Tokenizer, error) {
	switch d {
	case pack.KoToJa: // 한국어 제시 → 일본어로 작문
		t, err := tokenizer.New(ipa.Dict(), tokenizer.OmitBosEos())
		if err != nil {
			return nil, err
		}
		return &jaTokenizer{t: t}, nil
	case pack.JaToKo: // 일본어 제시 → 한국어로 작문
		t, err := tokenizer.New(koDict.Dict(), tokenizer.OmitBosEos())
		if err != nil {
			return nil, err
		}
		return &koTokenizer{t: t}, nil
	default:
		return nil, fmt.Errorf("알 수 없는 방향: %q", d)
	}
}

type jaTokenizer struct{ t *tokenizer.Tokenizer }

func (j *jaTokenizer) Tokenize(s string) []Token {
	raw := j.t.Tokenize(s)
	out := make([]Token, 0, len(raw))
	for _, r := range raw {
		tok := Token{
			Surface: r.Surface,
			Base:    r.Surface,
			Unknown: r.Class == tokenizer.UNKNOWN,
		}
		if pos := r.POS(); len(pos) > 0 {
			tok.POS = pos[0]
		}
		// IPADIC은 BaseForm을 제공한다.
		if b, ok := r.BaseForm(); ok && b != "*" && b != "" {
			tok.Base = b
		}
		out = append(out, tok)
	}
	return out
}

type koTokenizer struct{ t *tokenizer.Tokenizer }

func (k *koTokenizer) Tokenize(s string) []Token {
	raw := k.t.Tokenize(s)
	out := make([]Token, 0, len(raw))
	for _, r := range raw {
		f := r.Features()
		tok := Token{
			Surface: r.Surface,
			Base:    r.Surface,
			Unknown: r.Class == tokenizer.UNKNOWN,
		}
		if len(f) > 0 {
			tok.POS = f[0]
		}
		tok.Base = koreanBase(r.Surface, f)
		out = append(out, tok)
	}
	return out
}

// koreanBase는 mecab-ko-dic 토큰에서 어간 원형을 뽑는다.
//
// mecab-ko-dic은 BaseForm을 제공하지 않는다. 대신 활용형(Type=="Inflect")일 때
// Expression 필드에 형태소 분해가 들어 있다:
//
//	갔  VV+EP,*,T,갔,Inflect,VV,EP,가/VV/*+았/EP/*
//	                                ^^^ 어간 원형
//
// 첫 형태소의 "/" 앞부분이 어간이다. 활용형이 아니면 Surface가 곧 기본형이다.
func koreanBase(surface string, f []string) string {
	if len(f) <= koDict.Expression {
		return surface
	}
	if f[koDict.Type] != "Inflect" {
		return surface
	}
	expr := f[koDict.Expression]
	if expr == "" || expr == "*" {
		return surface
	}
	first := expr
	if i := strings.Index(first, "+"); i >= 0 {
		first = first[:i]
	}
	if i := strings.Index(first, "/"); i >= 0 {
		first = first[:i]
	}
	if first == "" {
		return surface
	}
	return first
}
```

```go
// internal/analyze/coverage.go
package analyze

import "strings"

// Coverage는 key_points 각 항목이 답안에 나타나는지 확인한다.
//
// 세 가지 방식으로 매칭한다:
//  1. 형태소의 기본형과 일치 (「思わなかった」→「思う」)
//  2. 형태소의 표층형과 일치
//  3. 표층형을 이어붙인 문자열에 포함 (「ていた」같은 연결 표현)
//
// 판정이 아니라 확인이다. missing이 곧 오답을 뜻하지 않는다.
func Coverage(tokens []Token, keyPoints []string) (covered, missing []string) {
	bases := make(map[string]bool, len(tokens))
	surfaces := make(map[string]bool, len(tokens))
	var joined strings.Builder
	for _, t := range tokens {
		bases[t.Base] = true
		surfaces[t.Surface] = true
		joined.WriteString(t.Surface)
	}
	flat := joined.String()

	covered = []string{}
	missing = []string{}
	for _, kp := range keyPoints {
		needle := stripTilde(kp)
		if needle == "" {
			continue
		}
		if bases[needle] || surfaces[needle] || strings.Contains(flat, needle) {
			covered = append(covered, kp)
		} else {
			missing = append(missing, kp)
		}
	}
	return covered, missing
}

// stripTilde는 key_point의 물결표를 제거한다. 「〜ていた」→「ていた」
func stripTilde(s string) string {
	s = strings.ReplaceAll(s, "〜", "")
	s = strings.ReplaceAll(s, "～", "")
	s = strings.ReplaceAll(s, "~", "")
	return strings.TrimSpace(s)
}

// UnknownTokens는 사전에 없는 토큰의 표층형을 모은다.
// 오타·조어·활용 오류의 강한 신호다. 다만 고유명사도 여기 걸리므로
// 오류 판정이 아니라 "확인해 보라"는 표시로만 쓴다.
func UnknownTokens(tokens []Token) []string {
	out := []string{}
	for _, t := range tokens {
		if t.Unknown {
			out = append(out, t.Surface)
		}
	}
	return out
}
```

```go
// internal/analyze/style.go
package analyze

import (
	"strings"

	"github.com/noel88/tsuzuri/internal/pack"
)

// DetectStyle은 답안의 문체를 추정한다.
// 판단 근거가 없으면 빈 문자열을 돌려준다 — 모른다고 말하는 것이
// 틀리게 단정하는 것보다 낫다.
func DetectStyle(tokens []Token) pack.Style {
	for _, t := range tokens {
		// 일본어: 정중체 표지
		if t.Base == "です" || t.Base == "ます" {
			return pack.StylePolite
		}
		// 일본어: 보통체 표지
		if t.Base == "だ" || t.Base == "である" {
			return pack.StylePlain
		}
		// 한국어: 종결어미로 판단
		if t.POS == "EF" {
			if isKoreanPolite(t.Surface) {
				return pack.StylePolite
			}
			return pack.StylePlain
		}
	}
	// 일본어 보통체는 명시적 표지 없이 「〜た」「〜る」로 끝나는 경우가 많다.
	// 정중체 표지가 하나도 없고 동사·형용사로 끝나면 보통체로 본다.
	if len(tokens) > 0 {
		last := tokens[len(tokens)-1]
		if last.POS == "助動詞" || last.POS == "動詞" || last.POS == "形容詞" {
			return pack.StylePlain
		}
	}
	return ""
}

func isKoreanPolite(surface string) bool {
	for _, suf := range []string{"요", "습니다", "ㅂ니다", "십시오", "세요"} {
		if strings.HasSuffix(surface, suf) {
			return true
		}
	}
	return false
}

// StyleMismatch는 답안의 문체가 문제가 요구한 문체와 어긋나는지 본다.
// 문체를 판단할 수 없으면 false다 — 확실할 때만 지적한다.
func StyleMismatch(tokens []Token, want pack.Style) bool {
	got := DetectStyle(tokens)
	if got == "" || want == "" {
		return false
	}
	return got != want
}
```

```go
// internal/analyze/vocab.go
package analyze

// VocabDiff는 답안과 참조의 내용어를 비교한다.
// 조사·어미는 제외한다 — 그쪽은 규칙으로 판정할 수 없는 영역이다.
//
// refOnly는 "참조에는 있는데 내 답에는 없는 것", ansOnly는 그 반대다.
// 어느 쪽도 오류를 뜻하지 않는다. 번역은 정답이 여럿이다.
func VocabDiff(answer, reference []Token) (refOnly, ansOnly []string) {
	ansSet := contentSet(answer)
	refSet := contentSet(reference)

	refOnly = []string{}
	for _, w := range contentWords(reference) {
		if !ansSet[w] {
			refOnly = append(refOnly, w)
		}
	}
	ansOnly = []string{}
	for _, w := range contentWords(answer) {
		if !refSet[w] {
			ansOnly = append(ansOnly, w)
		}
	}
	return refOnly, ansOnly
}

// contentWords는 내용어의 기본형을 등장 순서대로, 중복 없이 돌려준다.
func contentWords(ts []Token) []string {
	seen := map[string]bool{}
	out := []string{}
	for _, t := range ts {
		if !t.IsContent() || t.Base == "" {
			continue
		}
		if seen[t.Base] {
			continue
		}
		seen[t.Base] = true
		out = append(out, t.Base)
	}
	return out
}

func contentSet(ts []Token) map[string]bool {
	set := map[string]bool{}
	for _, w := range contentWords(ts) {
		set[w] = true
	}
	return set
}

// LenRatio는 참조 대비 답안의 형태소 개수 비율이다.
// 1.0에서 크게 벗어나면 지나치게 길거나 짧다는 신호다.
func LenRatio(answer, reference []Token) float64 {
	if len(reference) == 0 {
		return 0
	}
	return float64(len(answer)) / float64(len(reference))
}
```

```go
// internal/analyze/analyze.go
package analyze

import "github.com/noel88/tsuzuri/internal/pack"

// Analysis는 오프라인 분석의 결과다.
//
// 여기에 점수나 정오 판정은 없고, 앞으로도 추가하지 않는다 (스펙 §5.4).
// 참조 번역은 유일한 정답이 아니므로, 참조와 다르다는 이유로 감점하면
// 맞는 답을 틀렸다고 가르치게 된다. 이 구조체는 사실의 나열이다.
type Analysis struct {
	Covered       []string   `json:"covered"`
	Missing       []string   `json:"missing"`
	UnknownTokens []string   `json:"unknown_tokens"`
	StyleMismatch bool       `json:"style_mismatch"`
	DetectedStyle pack.Style `json:"detected_style,omitempty"`
	RefOnly       []string   `json:"ref_only"`
	AnsOnly       []string   `json:"ans_only"`
	LenRatio      float64    `json:"len_ratio"`
}

// Analyzer는 한 방향(=한 사전)에 대한 분석기다.
// 사전 로딩이 비싸므로 재사용한다.
type Analyzer struct {
	dir pack.Direction
	tk  Tokenizer
}

// New는 방향에 맞는 사전을 열어 분석기를 만든다.
// 사전 로딩이 여기서 일어난다 — 포메라에서는 수 초가 걸릴 수 있다.
func New(d pack.Direction) (*Analyzer, error) {
	tk, err := NewTokenizer(d)
	if err != nil {
		return nil, err
	}
	return &Analyzer{dir: d, tk: tk}, nil
}

// Analyze는 답안 하나를 분석한다.
func (a *Analyzer) Analyze(p pack.Problem, answer string) Analysis {
	ansTokens := a.tk.Tokenize(answer)
	refTokens := a.tk.Tokenize(p.Reference)

	covered, missing := Coverage(ansTokens, p.KeyPoints)
	refOnly, ansOnly := VocabDiff(ansTokens, refTokens)

	return Analysis{
		Covered:       covered,
		Missing:       missing,
		UnknownTokens: UnknownTokens(ansTokens),
		StyleMismatch: StyleMismatch(ansTokens, p.Style),
		DetectedStyle: DetectStyle(ansTokens),
		RefOnly:       refOnly,
		AnsOnly:       ansOnly,
		LenRatio:      LenRatio(ansTokens, refTokens),
	}
}
```

```go
// RenderProblem은 답안 입력 전 화면이다. 참조 번역을 노출하지 않는다.
func RenderProblem(p pack.Problem) string {
	var b strings.Builder
	b.WriteString(headerLine(fmt.Sprintf(" %s  %s  %s  %s ", p.ID, p.Dir, p.Level, p.Topic)))

// RenderResult는 제출 후 화면이다.
//
// 여기에는 점수도 정오 판정도 없다 (스펙 §5.4).
// ✓ ✗ ⚠ 는 오답 표시가 아니라 확인 요청이다.
func RenderResult(p pack.Problem, answer string, a analyze.Analysis, queueLen int, online bool) string {
	var b strings.Builder

	b.WriteString(headerLine(fmt.Sprintf(" %s  %s  %s  %s ", p.ID, p.Dir, p.Level, p.Topic)))
	b.WriteString("\n " + p.Prompt + "\n\n")
	b.WriteString(" 나:  " + answer + "\n")
	b.WriteString(" 참조: " + p.Reference + "\n\n")

	// 핵심 표현 — 있는 것과 빠진 것
	var marks []string
	for _, c := range a.Covered {
		marks = append(marks, "✓ 「"+c+"」")
	}
	for _, m := range a.Missing {
		marks = append(marks, "✗ 「"+m+"」 빠짐")
	}
	if len(marks) > 0 {
		b.WriteString(" " + strings.Join(marks, "    ") + "\n")
	}

	// 미지어 — 없으면 줄 자체를 만들지 않는다
	if len(a.UnknownTokens) > 0 {
		for _, u := range a.UnknownTokens {
			b.WriteString(" ⚠ 「" + u + "」 사전에 없는 형태 — 확인 필요\n")
		}
	}

	// 문체 — 어긋날 때만
	if a.StyleMismatch {
		b.WriteString(fmt.Sprintf(" ⚠ 문체  답안은 %s / 문제는 %s\n",
			styleLabel(a.DetectedStyle), styleLabel(p.Style)))
	}

	// 어휘 차이
	if len(a.RefOnly) > 0 || len(a.AnsOnly) > 0 {
		b.WriteString(" · ")
		if len(a.RefOnly) > 0 {
			b.WriteString("참조에만: " + strings.Join(a.RefOnly, ", ") + "   ")
		}
		if len(a.AnsOnly) > 0 {
			b.WriteString("내 답안에만: " + strings.Join(a.AnsOnly, ", "))
		}
		b.WriteString("\n")
	}

	b.WriteString("\n [Enter] 다음   [f] 첨삭 우선   [r] 다시 쓰기   [q] 종료\n")
	b.WriteString(footerLine(fmt.Sprintf(" 큐 %d건 · %s ", queueLen, connLabel(online))))
	b.WriteString("\n")
	return b.String()
}

	switch s {
	case pack.StylePolite:
		return "정중체"
	case pack.StylePlain:
		return "보통체"
	default:
		return "판단 불가"
	}
}

	if online {
		return "온라인"
	}
	return "오프라인"
}

	return "─" + title + strings.Repeat("─", max(0, width-runeLen(title)-1))
}

	return strings.Repeat("─", max(0, width-runeLen(status)-1)) + status + "─"
}

func runeLen(s string) int {
	return len([]rune(s))
}

func max(a, b int) int {
	if a > b {
		return a
	}
	return b
}
```

```go
// internal/ui/input.go
package ui

import (
	"bufio"
	"io"
	"os"
	"os/exec"
	"path/filepath"
	"strings"
)

// Command는 결과 화면에서의 다음 동작이다.
type Command int

const (
	CmdNext     Command = iota // Enter — 다음 문제
	CmdPriority                // f — 첨삭 우선 처리 표시
	CmdRetry                   // r — 같은 문제 다시 풀기
	CmdQuit                    // q — 종료
)

// ReadLine은 한 줄을 읽는다.
//
// raw mode를 쓰지 않는 것이 중요하다 (스펙 §6.1).
// 터미널의 정상 입력 경로를 그대로 써야 mozc IME가 동작한다.
func ReadLine(r io.Reader) (string, error) {
	br := bufio.NewReader(r)
	line, err := br.ReadString('\n')
	if err != nil && err != io.EOF {
		return "", err
	}
	// EOF로 끝나면 개행이 없다 — 읽은 내용은 유효하다.
	return strings.TrimRight(line, "\r\n"), nil
}

// ParseCommand는 결과 화면의 입력을 해석한다.
// 모르는 입력은 다음 문제로 넘어간다 — 흐름을 끊지 않는다.
func ParseCommand(s string) Command {
	switch strings.ToLower(strings.TrimSpace(s)) {
	case "f":
		return CmdPriority
	case "r":
		return CmdRetry
	case "q":
		return CmdQuit
	default:
		return CmdNext
	}
}

// IsEditorRequest는 답안 자리에 ":e"가 입력됐는지 본다.
func IsEditorRequest(s string) bool {
	return strings.TrimSpace(s) == ":e"
}

// ReadFromEditor는 임시 파일을 에디터로 열고, 저장된 내용을 돌려준다.
// 긴 답안을 쓸 때 쓴다.
func ReadFromEditor(editor, initial string) (string, error) {
	if editor == "" {
		editor = os.Getenv("EDITOR")
	}
	if editor == "" {
		editor = "nano"
	}

	dir, err := os.MkdirTemp("", "tsuzuri")
	if err != nil {
		return "", err
	}
	defer os.RemoveAll(dir)

	path := filepath.Join(dir, "answer.txt")
	if err := os.WriteFile(path, []byte(initial), 0o600); err != nil {
		return "", err
	}

	cmd := exec.Command(editor, path)
	cmd.Stdin = os.Stdin
	cmd.Stdout = os.Stdout
	cmd.Stderr = os.Stderr
	if err := cmd.Run(); err != nil {
		return "", err
	}

	b, err := os.ReadFile(path)
	if err != nil {
		return "", err
	}
	return strings.TrimSpace(string(b)), nil
}
```

```go
import (
	"bufio"
	"fmt"
	"io"
	"path/filepath"
	"strings"

	"github.com/noel88/tsuzuri/internal/analyze"

	"github.com/noel88/tsuzuri/internal/store"
	"github.com/noel88/tsuzuri/internal/ui"
)

// Session은 드릴 한 회차다. 한 방향만 다룬다.
type Session struct {
	Problems []pack.Problem
	Analyzer *analyze.Analyzer
	DataDir  string
	In       io.Reader
	Out      io.Writer
	Now      func() time.Time
}

// Run은 문제를 순서대로 출제하고, 답안과 분석 결과를 저장한다.
// 네트워크를 쓰지 않는다.
func (s *Session) Run() error {
	if s.Now == nil {
		s.Now = time.Now
	}
	br := bufio.NewReader(s.In)

	attemptsPath := filepath.Join(s.DataDir, "attempts.jsonl")
	queuePath := filepath.Join(s.DataDir, "queue.jsonl")

	existing, err := store.ReadAll[store.QueueItem](queuePath)
	if err != nil {
		return err
	}
	queueLen := len(existing)

	for i := 0; i < len(s.Problems); {
		p := s.Problems[i]

		fmt.Fprint(s.Out, ui.RenderProblem(p))

		answer, err := readAnswer(br)
		if err == io.EOF {
			return nil
		}
		if err != nil {
			return err
		}
		if strings.TrimSpace(answer) == "" {
			i++
			continue
		}

		a := s.Analyzer.Analyze(p, answer)

		at := s.Now()
		attempt := store.Attempt{
			ID:       fmt.Sprintf("a%d-%s", at.UnixNano(), p.ID),
			PackID:   p.ID,
			At:       at,
			Answer:   answer,
			Analysis: a,
		}
		if err := store.Append(attemptsPath, attempt); err != nil {
			return err
		}
		queueLen++

		fmt.Fprint(s.Out, ui.RenderResult(p, answer, a, queueLen, false))

		cmdLine, err := ui.ReadLine(br)
		if err != nil && err != io.EOF {
			return err
		}
		cmd := ui.ParseCommand(cmdLine)

		// 모든 답안을 큐에 적재한다. f는 우선 처리 표시일 뿐이다.
		item := store.QueueItem{
			AttemptID: attempt.ID,
			Priority:  cmd == ui.CmdPriority,
			At:        at,
		}
		if err := store.Append(queuePath, item); err != nil {
			return err
		}

		switch cmd {
		case ui.CmdQuit:
			return nil
		case ui.CmdRetry:
			// 같은 문제를 다시 낸다. 이전 attempt는 지우지 않는다.
			continue
		default:
			i++
		}
		if err == io.EOF {
			return nil
		}
	}
	return nil
}

// readAnswer는 답안 한 줄을 읽되, ":e"면 에디터를 연다.
func readAnswer(br *bufio.Reader) (string, error) {
	line, err := ui.ReadLine(br)
	if err != nil {
		return "", err
	}
	if ui.IsEditorRequest(line) {
		return ui.ReadFromEditor("", "")
	}
	return line, nil
}
```
